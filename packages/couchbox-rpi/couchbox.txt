[pi3]
dtoverlay=vc4-kms-v3d,cma-256

[pi02]
dtoverlay=vc4-kms-v3d,cma-256

[pi4]
dtoverlay=vc4-kms-v3d,cma-512

# warmer; fit a heatsink.

[pi5]
dtoverlay=vc4-kms-v3d,cma-512
